-- CAT HUB background image (uses the user's supplied asset ID).
-- Wait for Redz/Wand UI to create its ScreenGui, then place the image on the
-- main window frame so controls remain interactive above it.
task.spawn(function()
    local ok, err = pcall(function()
        local guiRoot = (gethui and gethui()) or game:GetService("CoreGui")
        local screenGui
        for _ = 1, 100 do
            screenGui = guiRoot:FindFirstChild("redz-library-v5")
            if screenGui then break end
            task.wait(0.1)
        end
        if not screenGui then return end

        task.wait(0.5)
        local rootFrame
        local largestArea = 0
        for _, obj in ipairs(screenGui:GetDescendants()) do
            if obj:IsA("Frame") and obj.Parent == screenGui then
                local size = obj.AbsoluteSize
                local area = size.X * size.Y
                if area > largestArea then
                    largestArea = area
                    rootFrame = obj
                end
            end
        end
        if not rootFrame then
            for _, obj in ipairs(screenGui:GetDescendants()) do
                if obj:IsA("Frame") then
                    local size = obj.AbsoluteSize
                    local area = size.X * size.Y
                    if area > largestArea then
                        largestArea = area
                        rootFrame = obj
                    end
                end
            end
        end
        if not rootFrame or rootFrame:FindFirstChild("CATHubBackground") then return end

        local background = Instance.new("ImageLabel")
        background.Name = "CATHubBackground"
        background.BackgroundTransparency = 1
        background.BorderSizePixel = 0
        background.Image = "rbxassetid://92251347397245"
        background.ImageTransparency = 0.35
        background.ScaleType = Enum.ScaleType.Crop
        background.Size = UDim2.fromScale(1, 1)
        background.Position = UDim2.fromScale(0, 0)
        background.AnchorPoint = Vector2.new(0, 0)
        background.ZIndex = rootFrame.ZIndex
        background.Parent = rootFrame
        rootFrame.BackgroundTransparency = 1
    end)
    if not ok then warn("CAT HUB background setup failed: " .. tostring(err)) end
end)

local function adaptContainer(container)
    if not container or container.__catHubAdapted then return container end
    container.__catHubAdapted = true

    local addSection = container.AddSection
    if addSection then
        container.AddSection = function(self, title)
            pcall(function() addSection(self, title) end)
            -- Redz sections are visual headings; return the tab so existing
            -- section-variable calls continue to add controls to that tab.
            return self
        end
    end

    local addToggle = container.AddToggle
    if addToggle then
        container.AddToggle = function(self, id, config)
            if type(id) == "table" then config, id = id, nil end
            config = config or {}
            config.Name = config.Name or config.Title or id or "Toggle"
            return addToggle(self, config)
        end
    end

    local addButton = container.AddButton
    if addButton then
        container.AddButton = function(self, id, config)
            if type(id) == "table" then config, id = id, nil end
            config = config or {}
            config.Name = config.Name or config.Title or id or "Button"
            return addButton(self, config)
        end
    end

    local addDropdown = container.AddDropdown
    if addDropdown then
        container.AddDropdown = function(self, id, config)
            if type(id) == "table" then config, id = id, nil end
            config = config or {}
            config.Name = config.Name or config.Title or id or "Dropdown"
            config.Options = config.Options or config.Values or {}
            local control = addDropdown(self, config)
            if control and not control.OnChanged then
                control.OnChanged = function(ctrl, callback)
                    if ctrl.AddCallback then ctrl:AddCallback(callback) end
                    return ctrl
                end
            end
            return control
        end
    end

    local addSlider = container.AddSlider
    if addSlider then
        container.AddSlider = function(self, id, config)
            if type(id) == "table" then config, id = id, nil end
            config = config or {}
            config.Name = config.Name or config.Title or id or "Slider"
            config.Increment = config.Increment or 1
            return addSlider(self, config)
        end
    end

    local addInput = container.AddInput or container.AddTextBox
    if addInput then
        container.AddInput = function(self, id, config)
            if type(id) == "table" then config, id = id, nil end
            config = config or {}
            config.Name = config.Name or config.Title or id or "Input"
            config.Placeholder = config.Placeholder or "Enter text"
            local control = addInput(self, config)
            return control
        end
    end

    local addParagraph = container.AddParagraph
    if addParagraph then
        container.AddParagraph = function(self, config, content)
            local title
            if type(config) == "table" then
                title = config.Title or "Info"
                content = config.Content or ""
            else
                title = tostring(config or "Info")
                content = tostring(content or "")
            end
            local control = addParagraph(self, title, content)
            if control and not control.SetDesc then
                control.SetDesc = function(ctrl, value)
                    if ctrl.SetDescription then return ctrl:SetDescription(tostring(value)) end
                    if ctrl.SetTitle then return ctrl:SetTitle(title .. "\n" .. tostring(value)) end
                end
            end
            return control
        end
    end

    return container
end

do
	local v15 = GRP_Info_Status_Server:AddParagraph({
		Title = "Time Zone",
		Content = ""
	})

	UpdateOS = function()
		local v16 = os.date("*t")
		local v17 = v16.hour % 24
		local v18 = string.format("%02i:%02i:%02i %s", (v17 - 1) % 12 + 1, v16.min, v16.sec, v17 < 12 and "AM" or "PM")
		local v19 = string.format("%02d/%02d/%04d", v16.day, v16.month, v16.year)
		local LocalizationService = game:GetService("LocalizationService")
		local LocalPlayer = game:GetService("Players").LocalPlayer
		local countryRegionCode

		if getgenv().countryRegionCode then
			countryRegionCode = getgenv().countryRegionCode
		else
			local ok

			ok, countryRegionCode = pcall(function()
				return LocalizationService:GetCountryRegionForPlayerAsync(LocalPlayer)
			end)

			if ok then
				getgenv().countryRegionCode = countryRegionCode
			else
				getgenv().countryRegionCode = "Unknown"
			end
		end

		v15:SetDesc(v19 .. " - " .. v18 .. " [ " .. countryRegionCode .. " ]")
	end
end

spawn(function()
	while true do
		UpdateOS()
		wait(1)
	end

rtxM = GRP_Misc_Graphics_Haki_Stats:AddToggle("Turn_on_RTX_Mode", {
	Title = "Turn on RTX Mode",
	Default = false,
	Callback = function(p267)
		_G.RTXMode = p267

		for _, child in pairs(game.Lighting:GetChildren()) do
			if child.Name == "RTXColorCorrection" or child.Name == "RTXTint" then
				child:Destroy()
			end
		end

		if _G.RTXMode then
			local ColorCorrectionEffect = Instance.new("ColorCorrectionEffect", game.Lighting)

			ColorCorrectionEffect.Name = "RTXColorCorrection"

			local ColorCorrectionEffect2 = Instance.new("ColorCorrectionEffect", game.Lighting)

			ColorCorrectionEffect2.Name = "RTXTint"
			OldAmbient = game.Lighting.Ambient
			OldBrightness = game.Lighting.Brightness
			OldColorShift_Top = game.Lighting.ColorShift_Top
			ColorCorrectionEffect.Brightness = 0.176
			ColorCorrectionEffect.Contrast = 0.39
			ColorCorrectionEffect.TintColor = Color3.fromRGB(217, 145, 57)
			ColorCorrectionEffect2.TintColor = Color3.fromRGB(217, 145, 57)
			game.Lighting.FogEnd = 999
			spawn(function()
				while _G.RTXMode do
					wait()
					game.Lighting.Ambient = Color3.fromRGB(33, 33, 33)
					game.Lighting.Brightness = 0.3

					if not _G.RTXMode then
						break
					end

					if plr.Character then
						if plr.Character:FindFirstChild("HumanoidRootPart") then
							if not plr.Character.HumanoidRootPart:FindFirstChild("RTXPointLight") then
								local PointLight = Instance.new("PointLight")

								PointLight.Name = "RTXPointLight"
								PointLight.Parent = plr.Character.HumanoidRootPart
								PointLight.Range = 15
								PointLight.Color = Color3.fromRGB(217, 145, 57)
							end
						end
					end
				end
			end)
		else
			pcall(function()
				if OldAmbient then
					game.Lighting.Ambient = OldAmbient
				end

				if OldBrightness then
					game.Lighting.Brightness = OldBrightness
				end

				if OldColorShift_Top then
					game.Lighting.ColorShift_Top = OldColorShift_Top
				end

				game.Lighting.FogEnd = 2500
			end)

			for _, child in pairs(game.Lighting:GetChildren()) do
				if child.Name == "RTXColorCorrection" or child.Name == "RTXTint" then
					child:Destroy()
				end
			end

			if plr.Character then
				if plr.Character:FindFirstChild("HumanoidRootPart") then
					local RTXPointLight = plr.Character.HumanoidRootPart:FindFirstChild("RTXPointLight")

					if RTXPointLight then
						RTXPointLight:Destroy()
					end
				end
			end
		end
	end
})
